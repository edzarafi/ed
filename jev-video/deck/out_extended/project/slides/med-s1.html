from typesafe_sdk import TypeSafeClient, Choice, Score, Noul

 

r = TypeSafeClient().system_one(

  state=document_text,

  questions={

    "doc_type": Choice(instructions="איזה סוג של מסמך רפואי זה?",

          criteria={"discharge_summary": "סיכום אשפוז...", "lab_report": ..., ...}),

    "specialty": Choice(criteria={"cardiology": None, "oncology": None, ...}),

    "urgency": Score(criteria=["לא נדרשת פעולה", "שבועות", "ימים", "24-48 שעות", "חירום"]),

    "contains_phi": Noul(instructions="האם המסמך מכיל שם או מספר זהות?"),

  })

 

r.choices["doc_type"].choice, r.choices["doc_type"].confidence # "imaging_report", 0.97
